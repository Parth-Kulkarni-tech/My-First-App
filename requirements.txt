import streamlit as st
from google import genai
from dotenv import load_dotenv
import time

load_dotenv()


# Initialize Gemini Client
client = genai.Client()
st.set_page_config(
    page_title="Parth's APP", 
    page_icon="✈️", # You can use an emoji or a path to an image file
    layout="wide"
)



# 2. Live Animated Heading using HTML/CSS
st.markdown(

    unsafe_allow_html=True,
)

# Divider for clean separation
st.divider()

st.caption("YOUR PERSONAL TRAVEL PLANNER")

# User Inputs
location = st.text_input("Where Do You Wanna Go ?")
days = st.number_input("How many days of trip?", min_value=1, max_value=30)
budget = st.selectbox("Select Budget", ["Luxury", "Moderate", "Budgeted"])
travel_type = st.radio("Who are you travelling with?", ["Solo", "Friends", "Family"])

# Prompt Construction
prompt = f"""You are a professional travel planner. The user wants to go to {location} for {days} days. 
Their budget is {budget}. Their travelling type is: {travel_type}.
Plan a detailed trip and share the answer in bullet-form."""

# Submission logic
if st.button("Plan Trip"):
    if not location.strip():
        st.warning("Please enter a destination first!")
    else:
        with st.status("✈️ Launching VoyageAI Planner...", expanded=True) as status:
            st.write(f"🔍 Analyzing destination: {location} for {days} days...")
            time.sleep(0.8)
            
            st.write(f"💰 Factoring in a {budget.lower()} budget framework...")
            time.sleep(0.8)
            
            st.write(f"🎒 Curating vibes for a {travel_type.lower()} adventure...")
            time.sleep(0.8)
            
            st.write("✨ Consulting global travel archives & generating itinerary...")
            response = client.models.generate_content(
                model="gemini-3.5-flash-lite",
                contents=prompt,
            )
        
        st.success("Woohoo! Here are some crazy suggestions...")
        st.write(response.text)
